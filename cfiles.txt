        printf("Invalid input.\n");
        return 1;
    }

    // Handle negative numbers
    if (n < 0) {
        printf("Factorial is not defined for negative numbers.\n");

        // Calculate factorial using a loop

            factorial *= i;
        }

    }

}

// Define Pi for degree-to-radian conversion
#define PI 3.14159265358979323846

int main() {
    int degrees;
    double radians;
    double sine, cosine, tangent, cotangent;

    // Get input from the user
    printf("Enter an integer angle in degrees: ");
    if (scanf("%d", &degrees) != 1) {
        printf("Invalid input.\n");
        return 1;
    }

    // Convert degrees to radians
    radians = degrees * (PI / 180.0);

    // Calculate trigonometric values
    sine = sin(radians);
    cosine = cos(radians);

    // Print the results with handling for undefined values (e.g., tan/cot at 90°, 270°, etc.)
    printf("\nResults for %d°:\n", degrees);
    printf("----------------------------------------\n");
    printf("Sine (sin)     : %.4f\n", sine);
    printf("Cosine (cos)   : %.4f\n", cosine);

    // Tangent is undefined where cosine is 0 (e.g., 90°, 270°)
    if (fabs(cosine) < 1e-9) {
        printf("Tangent (tan)  : Undefined (Infinite)\n");
    } else {
        tangent = tan(radians);
        printf("Tangent (tan)  : %.4f\n", tangent);
    }

    // Cotangent is undefined where sine is 0 (e.g., 0°, 180°, 360°)
    if (fabs(sine) < 1e-9) {
        printf("Cotangent (cot): Undefined (Infinite)\n");
    } else {
        cotangent = 1.0 / tan(radians);
        printf("Cotangent (cot): %.4f\n", cotangent);
    }
    printf("----------------------------------------\n");

    return 0;
}

#define N 3

// Function to display a 3x3 matrix
void displayMatrix(float mat[N][N]) {
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            printf("%8.3f ", mat[i][j]);
        }
        printf("\n");
    }
}

// Function to find the inverse of matrix B using Gauss-Jordan elimination
int inverseMatrix(float b[N][N], float inv[N][N]) {
    float temp[N][2 * N];

    // Initialize augmented matrix [B | I]
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            temp[i][j] = b[i][j];
            temp[i][j + N] = (i == j) ? 1.0 : 0.0;
        }
    }

    // Apply Gauss-Jordan elimination
    for (int i = 0; i < N; i++) {
        // Pivot element check
        if (fabs(temp[i][i]) < 1e-9) {
            int swap_row = -1;
            for (int k = i + 1; k < N; k++) {
                if (fabs(temp[k][i]) > 1e-9) {
                    swap_row = k;
                    break;
                }
            }
            if (swap_row == -1) {
                return 0; // Matrix is singular (non-invertible)
            }
            // Swap rows
            for (int j = 0; j < 2 * N; j++) {
                float t = temp[i][j];
                temp[i][j] = temp[swap_row][j];
                temp[swap_row][j] = t;
            }
        }

        // Make the diagonal element 1
        float pivot = temp[i][i];
        for (int j = 0; j < 2 * N; j++) {
            temp[i][j] /= pivot;
        }

        // Make other elements in the current column 0
        for (int k = 0; k < N; k++) {
            if (k != i) {
                float factor = temp[k][i];
                for (int j = 0; j < 2 * N; j++) {
                    temp[k][j] -= factor * temp[i][j];
                }
            }
        }
    }

    // Extract the inverse matrix from the augmented matrix
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            inv[i][j] = temp[i][j + N];
        }
    }
    return 1;
}

// Function to multiply two 3x3 matrices: Result = A x B_inv
void multiplyMatrices(float a[N][N], float b_inv[N][N], float res[N][N]) {
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            res[i][j] = 0;
            for (int k = 0; k < N; k++) {
                res[i][j] += a[i][k] * b_inv[k][j];
            }
        }
    }
}

int main() {
    float A[N][N], B[N][N], B_inv[N][N], Result[N][N];

    printf("Enter elements of Matrix A (3x3):\n");
    for (int i = 0; i < N; i++) {

    }

    multiplyMatrices(A, B_inv, Result);

    displayMatrix(Result);

    return 0;
}

#define N 3 // Number of variables

// Function to perform Gaussian Elimination
int solveGaussianElimination(float mat[N][N + 1], float res[N]) {
    int i, j, k;
    for (i = 0; i < N; i++) {
        // Search for maximum element in the current column
        int maxRow = i;
        for (k = i + 1; k < N; k++) {
            if (mat[k][i] > mat[maxRow][i]) {
                maxRow = k;
            }
        }

        // Swap the maximum row with the current row
        for (k = i; k <= N; k++) {
            float temp = mat[i][k];
            mat[i][k] = mat[maxRow][k];
            mat[maxRow][k] = temp;
        }

        // Check if matrix is singular
        if (mat[i][i] == 0) {
            return -1; // Singular Matrix (Infinite solutions or no solution)
        }

        // Make the rows below this one zero in the current column
        for (k = i + 1; k < N; k++) {
            float factor = mat[k][i] / mat[i][i];
            for (j = i; j <= N; j++) {
                mat[k][j] -= factor * mat[i][j];
            }
        }
    }

    // Back-substitution to find x, y, z
    for (i = N - 1; i >= 0; i--) {
        res[i] = mat[i][N];
        for (j = i + 1; j < N; j++) {
            res[i] -= mat[i][j] * res[j];
        }
        res[i] = res[i] / mat[i][i];
    }
    return 1;
}

int main() {
    float a1, b1, c1, a2, b2, c2;
    float determinant, x, y;

    // Input the coefficients for the two equations
    printf("Enter coefficients for the first equation (a1x + b1y = c1):\n");
    printf("Enter a1, b1, c1: ");
    scanf("%f %f %f", &a1, &b1, &c1);

    printf("\nEnter coefficients for the second equation (a2x + b2y = c2):\n");
    printf("Enter a2, b2, c2: ");
    scanf("%f %f %f", &a2, &b2, &c2);

    // Calculate the main determinant
    determinant = (a1 * b2) - (a2 * b1);

    // Check for solutions based on the determinant
    if (determinant == 0) {
        // Check if dependent (infinitely many solutions) or inconsistent (no solution)
        // Using cross-multiplication ratios
        if ((a1 * c2 == a2 * c1) && (b1 * c2 == b2 * c1)) {
            printf("\nResult: The system has infinitely many solutions (dependent equations).\n");
        } else {
            printf("\nResult: The system has no solution (parallel lines).\n");
        }
    } else {
        // Using Cramer's Rule to find x and y
        x = ((c1 * b2) - (c2 * b1)) / determinant;
        y = ((a1 * c2) - (a2 * c1)) / determinant;

        printf("\nResult: The system has a unique solution.\n");
        printf("x = %.2f\n", x);
        printf("y = %.2f\n", y);
    }

    return 0;
}

#define MAX_DIGITS 100

// Function to add two large numbers represented as strings
void addLargeNumbers(char num1[], char num2[]) {
    int len1 = strlen(num1);
    int len2 = strlen(num2);
    
    // Result array size will be max length of operands + 1 for possible carry
    int max_len = (len1 > len2) ? len1 : len2;
    char *result = (char *)malloc(max_len + 2);
    
    int i = len1 - 1;
    int j = len2 - 1;
    int k = 0;
    int carry = 0;
    
    // Loop from the rightmost digits to the leftmost digits
    while (i >= 0 || j >= 0 || carry > 0) {
        int sum = carry;
        
        if (i >= 0) {
            sum += num1[i] - '0'; // Convert char to int
            i--;
        }
        
        if (j >= 0) {
            sum += num2[j] - '0'; // Convert char to int
            j--;
        }
        
        // Calculate the new carry and store the last digit of sum
        carry = sum / 10;
        result[k++] = (sum % 10) + '0'; // Convert int back to char
    }
    
    result[k] = '\0'; // Null-terminate the result string
    
    // Since the result was built backwards, reverse it for final output
    printf("Sum: ");
    for (int idx = k - 1; idx >= 0; idx--) {
        printf("%c", result[idx]);
    }
    printf("\n");
    
    free(result); // Free dynamically allocated memory
}

    scanf("%s", num1);

    scanf("%s", num2);
    
    addLargeNumbers(num1, num2);
    
    return 0;
}

int main() {
    double principal, rate, time, amount, compound_interest;

    // Input principal, rate, and time from the user
    printf("Enter principal amount (P): ");
    scanf("%lf", &principal);

    printf("Enter annual interest rate in percentage (R): ");
    scanf("%lf", &rate);

    printf("Enter time period in years (t): ");
    scanf("%lf", &time);

    // Calculate total amount using the formula: A = P * (1 + R/100)^t
    amount = principal * pow((1 + rate / 100), time);

    // Calculate compound interest
    compound_interest = amount - principal;

    // Display the results
    printf("\n--- Results ---\n");
    printf("Total Amount (A): %.2lf\n", amount);
    printf("Compound Interest (CI): %.2lf\n", compound_interest);

    return 0;
}
